\begin{enumerate}[roomy]


\item Consider the integers $\Z$ equipped with the finite complement topology. Compare the product topology on $\Z \times \Z$ with the finite complement topology on $\Z \times \Z$: determine if they are equal, and if not, if one is finer than the other, or if they are incomparable.



\item Consider the usual order on $\N$. Show that the dictionary topology on $\N \times \N$ is not the discrete topology.


\item Let $(X, \T_X)$ and $(Y, \T_Y)$ be topological spaces. 


\begin{enumerate}
\item Show that if $A$ is closed in $X$ and $B$ is closed in $Y$, then $A \times B$ is closed in $X \times Y$.


\item Show that if $A$ and $B$ are nonempty, and $A \times B$ is closed in $X \times Y$, then $A$ is closed in $X$.

\noindent
Note that analogously, we can conclude that $B$ is closed in $Y$.
\end{enumerate} 


\end{enumerate}
