\chead{Problem Set 11}



\begin{enumerate}[roomy]
\item Let $\T_1$ and $\T_2$ be topologies on $X$ such that both $(X, \T_1)$ and $(X, \T_2)$ are compact Hausdorff spaces. Show that if there exists $U \in \T_1$ such that $U \notin \T_2$, then there exists $V \in \T_2$ such that $V \notin \T_1$.


\item Let $A$ and $B$ be disjoint compact subspaces of the Hausdorff space $X$.
Show that there exist disjoint open sets $U$ and $V$ such that $A \subseteq U$ and $B \subseteq V$. 



\end{enumerate}
