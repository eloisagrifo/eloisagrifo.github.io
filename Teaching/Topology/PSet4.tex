\begin{enumerate}[roomy]


\item Find the closure of $(0,1)$ in $\R_{\ell}$.


\item Let $X$ be a topological space and consider $A \subseteq B \subseteq X$ with the subspace topologies.

\begin{enumerate}
	\item Show that if $A$ is dense in $B$ and $B$ is dense in $X$, then $A$ is dense in $X$.

\item Show that if $A$ is not dense in $B$, then $A$ is not dense in $X$.
\end{enumerate}


\item Let $X$ and $Y$ be topological spaces and consider $A \subseteq X$ and $B \subseteq Y$.
Show that in $X \times Y$ with the product topology,
\[ \overline{A \times B} = \overline{A} \times \overline{B}.\]



\item Let $X$ be a topological space and $A \subseteq X$. The {\bf boundary} of $A$ is defined by
\[ \bd(A) = \overline{A} \cap \overline{X \setminus A}.\]

\begin{enumerate}

\item Show that $A\inte$ and $\bd(A)$ are disjoint.

\item Show that $\overline{A} = A\inte \cup \bd(A)$.

\item Show that $\bd(A) = \varnothing$ if and only if $A$ is a clopen set.
\end{enumerate}


\end{enumerate}
