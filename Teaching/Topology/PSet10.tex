\chead{Problem Set 10}



\begin{enumerate}[roomy]
\item Let $A$ and $B$ nonempty open subsets of a topological space $Z$. 
Prove that if both $A \cap B$ and $A \cup B$ are connected, then each of $A$ and $B$ is connected.

\item Let $X$ be a connected space and endow $\R$ with the standard topology and $[0,1] \subseteq \R$ with the subspace topology.
Let $f,g \colon X \longrightarrow [0,1]$ be continuous functions and assume that $f$ is surjective. 
Show that there exists $x \in X$ with $f(x) = g(x)$.


\item Let $X$ and $Y$ be topological spaces and $f \colon X \longrightarrow Y$ a continuous function. Define an equivalence relation on $X$ by $x_1 \sim x_2$ if and only if $f(x_1) = f(x_2)$. 
Show that if $X$ is connected and $Y$ is Hausdorff, then the quotient space $X/\sim$ with the quotient topology is both connected and Hausdorff.

\end{enumerate}
