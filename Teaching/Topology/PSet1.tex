\begin{enumerate}[roomy]


\item Let $X$ be a set. Set
\[ \T = \{U \subseteq X \mid X \setminus U \text{ is infinite}\} \cup \{\emptyset, X\}.\] 
Show that $\T$ is a topology on $X$ if and only if $X$ is finite.	
	

\item Prove or disprove: is
	\[ \T = \{(-\infty, a] \mid a \in \R\} \cup \{\emptyset, \R\}\]
	a topology on $\R$?
	

\item Let $\{ \T_{\alpha} \}_{\alpha}$ be a collection of topologies on a set $X$.

\begin{enumerate}
\item Prove or disprove: is $\displaystyle\bigcap_\alpha \T_{\alpha}$ always a topology on $X$?
\item Prove or disprove: is $\displaystyle\bigcup_\alpha \T_{\alpha}$ always a topology on $X$?

\noindent
Either show that it is always a topology or given an example where it is not, with proof.
\end{enumerate}
\end{enumerate}
