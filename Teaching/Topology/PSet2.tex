\begin{enumerate}[roomy]


\item Show that the topologies $\R_{\ell}$ and $\R_K$ on $\R$ are incomparable.

\item Let $\B$ be a basis for a topology on $X$ and let $\T$ be the topology generated by $\B$. Let $\{\T_\alpha \}_{\alpha \in I}$ be the collection of all topologies on $X$ that contain $\B$.
Show that
\[ \T = \bigcap_{\alpha \in I} \T_\alpha.\]


\item Let $\B_1$ and $\B_2$ be bases for a topology on $X$.


\begin{enumerate}
	\item Show that the set
\[ \B := \{B_1 \cap B_2 \mid B_1 \in \B_1 \text{ and } B_2 \in \B_2 \}\]
is also a basis for a topology on $X$.


\item Show that the topology $\T$ generated by $\B$ is the coarsest topology on $X$ containing both $\B_1$ and $\B_2$, meaning that all topologies $\T'$ with $\B_1 \subseteq \T'$ and $\B_2 \subseteq \T'$ satisfy $T \subseteq T'$.
\end{enumerate}

\end{enumerate}
