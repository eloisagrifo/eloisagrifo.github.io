\begin{enumerate}[roomy]

\item Let $X$ and $Y$ be topological spaces and $f \colon X \to Y$ be an injective continuous function.

\begin{enumerate}
\item Prove that if $Y$ is Hausdorff then $X$ is Hausdorff.
\item Does the converse of (a) hold?
\end{enumerate}


\item Let $X$ be a Hausdorff space. 

\begin{enumerate}
\item Show that if $f \colon X \to X$ is continuous, then 
\[ C := \{ x \in X  \mid f(x) = x \}\]
is closed.

\item Let $A \subseteq X$ have the subspace topology. 
Show that if there exists a retraction $r \colon X \longrightarrow A$ of $X$ onto $A$, then $A$ is a closed subset of $X$.

\item Does (b) hold if we do not assume that $X$ is Hausdorff?
\end{enumerate}



\end{enumerate}
