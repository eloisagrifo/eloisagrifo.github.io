\begin{enumerate}[roomy]

\item Consider the box topology on $\R^{\omega}$, and let $A$ be the subset of $\R^{\omega}$ consisting of all the sequences whose terms are strictly positive:
\[A = \{ x = (x_n)_{n \geqslant 1} \in \R^{\omega} \mid x_n > 0 \text{ for all } n \geqslant 1 \}.\]

\begin{enumerate}
\item Let $z = (0)_{n \geqslant 1} \in \R^{\omega}$ be the constant sequence equal to zero. Show that $z \in \overline{A}$.

\item Show that there is no sequence of elements of $A$ converging to $z$.

Hint: given any sequence $(a_n)_{n \geqslant 1}$ of elements of $A$, let
\[ a_n = (x_{n,1},x_{n,2},x_{n,3}, \ldots ) = (x_{n,i})_{i \geqslant 1}\]
and consider the open set
\[ U = \prod_{n =1}^\infty (-x_{n,n}, x_{n,n}).\]

\item Conclude that $\R^{\omega}$ with the box topology is not metrizable.
\end{enumerate}


%We showed in class that the box topology on $\R^I$ is always finer than the uniform topology for any set $I$. In this next problem, we will show that the box topology is strictly finer than the uniform topology on $\R^{\omega}$.

\item Let $\overline{\rho}$ be the uniform metric on $\R^{\omega}$.
Given $x = (x_n)_{n \geqslant 1} \in \R^{\omega}$ and $0 < \e \leqslant 1$, let
\[ U(x, \e) = \prod_{n = 1}^\infty (x_n-\e, x_n + \e).\]
Let $z = (0)_{n \geqslant 1} \in \R^{\omega}$ be the constant sequence equal to zero. 
Show that $U(z,1)$ is not open in the uniform topology.

\end{enumerate}
